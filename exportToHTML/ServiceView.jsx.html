    useEffect(() => {
        const wrap = wrapRef.current;
        if (!wrap) return;

        const total = serviceList.length;
        const lastIndex = total - 1;

        let currentIndex = 0;
        let snapTimer = null;
        let ticking = false;

        const metrics = () => {
            const sticky = wrap.querySelector('.service-section');
            const stickyTop = parseFloat(getComputedStyle(sticky).top) || 0;

            // 카드 전환은 sticky가 실제로 고정되는 시점부터 시작
            const start = wrap.getBoundingClientRect().top + window.scrollY - stickyTop;
            const end = start + wrap.offsetHeight - sticky.offsetHeight;

            return { start, end: Math.max(start, end) };
        };

        const updateActiveIndex = () => {
            if (window.innerWidth <= 960) return;

            const { start, end } = metrics();

            if (end <= start) return;

            const y = window.scrollY;

            /* 
             * service-scroll-wrap 영역을 완전히 벗어난 경우 
             * 내부 카드 제어를 종료한다. 
             */
            if (y < start || y > end) {
                return;
            }

            const progress = Math.min(
                Math.max(
                    (y - start) / (end - start),
                    0
                ),
                1
            );

            const nextIndex = Math.max(
                0,
                Math.min(
                    lastIndex,
                    Math.round(progress * lastIndex)
                )
            );

            if (nextIndex === currentIndex) return;

            currentIndex = nextIndex;
            setActiveIndex(nextIndex);
        };

        /* 
         * 현재 스크롤 위치에서 가장 가까운 카드 위치로 
         * 짧게 스냅한다. 
         * 
         * wheel 이벤트를 막지 않기 때문에 
         * 터치패드의 연속적인 스크롤도 그대로 Lenis가 처리한다. 
         */
        const snapToNearestCard = () => {
            if (window.innerWidth <= 960) return;
            if (!window.lenis) return;

            const { start, end } = metrics();

            if (end <= start) return;

            const y = window.scrollY;

            /* 
             * service 영역 밖에서는 절대 snap하지 않는다. 
             * → 마지막 카드 이후 아래로 빠져나갈 때 
             *   페이지 전체 스크롤을 그대로 허용한다. 
             */
            if (y < start || y > end) return;

            const progress = Math.min(
                Math.max(
                    (y - start) / (end - start),
                    0
                ),
                1
            );

            const nextIndex = Math.max(
                0,
                Math.min(
                    lastIndex,
                    Math.round(progress * lastIndex)
                )
            );

            const target =
                start +
                (nextIndex / lastIndex) *
                (end - start);

            /* 
             * 이미 거의 정확한 위치라면 
             * 불필요한 Lenis 애니메이션을 실행하지 않는다. 
             */
            if (Math.abs(y - target) < 10) {
                currentIndex = nextIndex;
                setActiveIndex(nextIndex);
                return;
            }

            currentIndex = nextIndex;
            setActiveIndex(nextIndex);

            window.lenis.scrollTo(target, {
                duration: 0.25,
                force: false,
            });
        };

        /* 
         * Lenis가 처리한 실제 스크롤 위치를 기준으로 
         * 현재 카드만 변경한다. 
         */
        const onScroll = () => {
            if (window.innerWidth <= 960) return;

            if (!ticking) {
                window.requestAnimationFrame(() => {
                    updateActiveIndex();
                    ticking = false;
                });

                ticking = true;
            }

            /* 
             * 스크롤 입력이 잠시 멈추면 
             * 가장 가까운 카드 위치로 가볍게 정렬한다. 
             * 
             * debounce 방식이라 터치패드의 연속 입력을 
             * 중간중간 강제로 끊지 않는다. 
             */
            clearTimeout(snapTimer);

            snapTimer = setTimeout(() => {
                snapToNearestCard();
            }, 100);
        };

        /* 
         * 초기 진입 시 현재 위치에 맞는 카드 계산 
         */
        updateActiveIndex();

        window.addEventListener(
            'scroll',
            onScroll,
            {
                passive: true,
            }
        );

        window.addEventListener(
            'resize',
            updateActiveIndex
        );

        return () => {
            window.removeEventListener(
                'scroll',
                onScroll
            );

            window.removeEventListener(
                'resize',
                updateActiveIndex
            );

            clearTimeout(snapTimer);
        };
    }, []);

    // 모바일 전용 Sticky 처리
    useEffect(() => {

        // nav의 실제 하단 위치를 그대로 사용 (전환 중에도 현재 값 반영)
        const measure = () => {
            const stick = nav
                ? Math.round(nav.getBoundingClientRect().bottom) - 1
                : 0;

            wrap.style.setProperty(
                '--stick-top',
                `${stick}px`
            );

            wrap.style.setProperty(
                '--cards-top',
                `${stick + img.offsetHeight}px`
            );
        };

        let ticking = false;
        const update = () => {
            ticking = false;
            if (window.innerWidth > 960) return;
            measure();
            const line = img.getBoundingClientRect().bottom + 80;
            let idx = 0;
            cards.forEach((c, i) => {
                if (c.getBoundingClientRect().top <= line) idx = i;
            });
            setActiveIndex(idx);
        };
        const onScroll = () => {
            if (!ticking) { ticking = true; requestAnimationFrame(update); }
        };

        window.addEventListener('scroll', onScroll, { passive: true });
        window.addEventListener('resize', onScroll);
        nav?.addEventListener('transitionend', measure);   // nav 전환이 끝난 뒤 한 번 더 보정
        update();
        return () => {
            window.removeEventListener('scroll', onScroll);
            window.removeEventListener('resize', onScroll);
            nav?.removeEventListener('transitionend', measure);
        };
    }, []);

                            </div>
